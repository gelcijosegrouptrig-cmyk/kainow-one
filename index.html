import os import json from http.server import BaseHTTPRequestHandler, HTTPServer from github import Github from openai import OpenAI # Coleta o token automático e seguro do ambiente token = os.getenv("GITHUB_TOKEN") or os.getenv("CODESPACE_VSCODE_FOLDER") client = OpenAI( base_url="https://azure.com", api_key=token ) g = Github(token) repo = g.get_repo("gelcijosegrouptrig-cmyk/kainow-one") # HTML PREMIUM incorporado diretamente no servidor Python (Sem depender de arquivos externos) HTML_INTERFACE = """ Kainow IA Bot

🤖 Kainow Full-Stack IA

Online (Python Engine)
Olá! Eu sou o seu Agente Full-Stack rodando direto em Python. Digite abaixo o sistema, tela ou integração que você deseja criar!
""" class ServidorCompleto(BaseHTTPRequestHandler): def do_GET(self): # Serve a interface HTML escura instantaneamente sem cache self.send_response(200) self.send_header('Content-Type', 'text/html; charset=utf-8') self.send_header('Cache-Control', 'no-store, no-cache, must-revalidate') self.end_headers() self.wfile.write(HTML_INTERFACE.encode('utf-8')) def do_POST(self): if self.path == '/api/criar': content_length = int(self.headers['Content-Length']) post_data = self.rfile.read(content_length) dados = json.loads(post_data.decode('utf-8')) prompt = dados.get('prompt') print(f"🤖 [PYTHON ENGINE] Processando instrução de código: {prompt}") # OpenAI chama o modelo GPT-4o response = client.chat.completions.create( messages=[ {"role": "system", "content": "Você é um programador Full-Stack sênior experiente. Escreva estritamente APENAS o código puro, limpo e funcional solicitado (HTML/CSS/JS). Nunca inclua introduções, explicações textuais ou blocos de marcação markdown."}, {"role": "user", "content": prompt} ], model="gpt-4o", temperature=0.2 ) codigo_gerado = response.choices.message.content.strip() codigo_gerado = codigo_gerado.replace("```html", "").replace("```", "").strip() nome_arquivo = f"sistema-{int(os.urandom(2).hex(), 16)}.html" # Envia direto para o repositório repo.create_file( path=nome_arquivo, message=f"🤖 Bot IA: Modulo gerado via Python Engine", content=codigo_gerado, branch="main" ) self.send_response(200) self.send_header('Content-Type', 'application/json') self.end_headers() resposta = {"status": "sucesso", "arquivo": nome_arquivo} self.wfile.write(json.dumps(resposta).encode('utf-8')) print("🔥 [PYTHON SUPER ENGINE] Servidor 100% ativo rodando na porta 5000!") server = HTTPServer(('0.0.0.0', 5000), ServidorCompleto) server.serve_forever()
